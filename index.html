計算・探索の説明

評価式の全体像

\[D_{\mathrm{total}}=\sum_p D_p,\qquad D_p=T_p\left(\mathrm{DPS}_{\mathrm{N},p}+\mathrm{DPS}_{\mathrm{AS},p}+\mathrm{DPS}_{\mathrm{ULT},p}+\mathrm{DPS}_{\mathrm{ES},p}\right)\]

① EXP・覚醒Lv・基礎攻撃力

\[ \mathrm{EXP}^{\mathrm{eff}}_p = \mathrm{EXP}_p \left(1+\sum_{\mathrm{SE}}G_{\mathrm{SE}}\right) \left( (1-a) + a(1+A_{1,\mathrm{EXP},p})(1+A_{2,\mathrm{EXP},p}) (1+P_{\mathrm{EXP},p}\,\mathrm{CD}_{\mathrm{EXP},p}) \right) \]
\[ \mathrm{ALv}_p=F_{\mathrm{level}}\!\left(\mathrm{EXP}^{\mathrm{eff}}_p\right) \]
\[ \mathrm{ATK}_{\mathrm{base},p} = \mathrm{ATK}_0 F_{\mathrm{CR}}(\mathrm{CR}) F_{\mathrm{WR}}(\mathrm{WR}) F_{\mathrm{UM}}(\mathrm{UM}) F_{\mathrm{ALv}}(\mathrm{ALv}_p) \]

② 攻撃力補正

\[\mathrm{ATK}=\mathrm{ATK}_{\mathrm{base}}(1+A_1)(1+A_2)\]
\[A_1=\sum_{\mathrm{SE}}A_{\mathrm{SE}}\]
\[A_2=\sum_{\mathrm{Skill}}A_{\mathrm{Skill}}+A_{\mathrm{Ext}}+A_{\mathrm{Qtz}}+A_{\mathrm{Crys}}+A_{\mathrm{Back}}\]

③ 部位補正・弱点命中率

\[w_i=\frac{h_i}{\sum_jh_j},\qquad C_{\mathrm{part}}=\sum_iw_iC_i\]
\[C_i=\begin{cases}(1+B_i+W_x)(1+W_{\mathrm{weapon}})&\text{弱点部位}\\1-R_i&\text{Resist部位}\\1&\text{非弱点部位}\end{cases},\qquad P_{\mathrm{weak}}=\sum_{i\in\mathrm{weak}}w_i\]

④ 属性補正・デバフ補正

\[C_{\mathrm{elem}}=\begin{cases}1&\text{等倍}\\1+\min(E,1)&\text{属性弱点}\\1-R_{\mathrm{elem}}&\text{属性耐性}\end{cases}\]
\[C_{\mathrm{debuff}}=1+D\]

⑤ クリティカル率・クリティカルダメージージ期待値

\[P=\min\!\left(P_0+\sum_{\mathrm{SE}}P_{\mathrm{SE}}+\sum_{\mathrm{Skill}}P_{\mathrm{Skill}}+P_{\mathrm{BC}}+P_{\mathrm{Ext}},1\right)\]
\[C_{\mathrm{crit}}=1+P\cdot\mathrm{CD},\qquad \mathrm{CD}=\mathrm{CD}_0+\sum_{\mathrm{SE}}\mathrm{CD}_{\mathrm{SE}}+\sum_{\mathrm{Skill}}\mathrm{CD}_{\mathrm{Skill}}+\mathrm{CD}_{\mathrm{Ext}}\]

⑦ 敵ダウン・クワイエタス

\[T_{\mathrm{down},p}=N_{\mathrm{down},p}t_{\mathrm{down}},\qquad U_{\mathrm{Quietus},p}=\min\!\left(1,\frac{T_{\mathrm{down},p}}{T_p}\right)\]
\[B^{\mathrm{eff}}_{\mathrm{Quietus},x,p}=B_{\mathrm{Quietus}}\,U_{\mathrm{Quietus},p}\]

⑧ 敵攻撃周期とガード・PD・PG戦略探索

\[f_{\mathrm{enemy}}=1/\tau_{\mathrm{enemy}},\qquad a_p\in\{\mathrm{Guard},\mathrm{PD},\mathrm{PG}\}\]
\[P_{\mathrm{PDhit}}=(1-S_{\mathrm{PD}})H_{\mathrm{PD}},\quad P_{\mathrm{PGhit}}=(1-S_{\mathrm{PG}})H_{\mathrm{PG}},\quad P_{\mathrm{PG}\to\mathrm{Guard}}=(1-S_{\mathrm{PG}})(1-H_{\mathrm{PG}})\]

⑨ ガード・PD・PGによる攻撃可能時間損失

\[N_{\mathrm{PD}}=\frac{T_{\mathrm{base}}}{\tau_{\mathrm{PD}}},\quad L_{\mathrm{PD}}=\min(T_{\mathrm{base}},N_{\mathrm{PD}}t_{\mathrm{PDloss}})\]
\[L_{\mathrm{PDhit}}=\min(T_{\mathrm{base}}-L_{\mathrm{PD}},N_{\mathrm{PD}}P_{\mathrm{PDhit}}t_{\mathrm{PDrec}})\]
\[N_{\mathrm{G}}=T_{\mathrm{base}}f_{\mathrm{enemy}}\mathbf1_{a_p=\mathrm{Guard}},\quad L_{\mathrm{G}}=\min(T_{\mathrm{base}},N_{\mathrm{G}}t_{\mathrm{G,loss}})\]\n
\[N_{\mathrm{PG}}=T_{\mathrm{base}}f_{\mathrm{enemy}}\mathbf1_{a_p=\mathrm{PG}},\quad L_{\mathrm{PG}}=\min(T_{\mathrm{base}}-L_{\mathrm{G}},N_{\mathrm{PG}}t_{\mathrm{PG,loss}})\]
\[L_{\mathrm{PGhit}}=\min(T_{\mathrm{base}}-L_{\mathrm{G}}-L_{\mathrm{PG}},N_{\mathrm{PG}}P_{\mathrm{PGhit}}t_{\mathrm{PGrec}})\]
\[T_{\mathrm{attack}}=\max\!\left(0,T_{\mathrm{base}}-\min(T_{\mathrm{base}},L_{\mathrm{G}}+L_{\mathrm{PD}}+L_{\mathrm{PDhit}}+L_{\mathrm{PG}}+L_{\mathrm{PGhit}})\right)\]

⑩ HP最大維持率

\[\mathcal C(f,s,L)=1-\left[(1-q)(1-s)^n+q(1-s)^{n+1}\right]\]
\[T=1/f,\qquad n=\left\lfloor\frac LT\right\rfloor,\qquad q=\frac{L-nT}{T}\]
\[C_{\mathrm{PD}}=\mathcal C(f_{\mathrm{PD}},P_{\mathrm{PDhit}},L_{\mathrm{HP},\mathrm{PD}}),\qquad C_{\mathrm{PG}}=\mathcal C(f_{\mathrm{PG}},P_{\mathrm{PGhit}},L_{\mathrm{HP},\mathrm{PG}})\]
\[C_{\mathrm{HP}-\mathrm{loss}}=1-(1-C_{\mathrm{PD}})(1-C_{\mathrm{PG}})\]
\[U_{\mathrm{HPmax}}=\begin{cases}0&\text{通常ガード成立回数}>0\\U_{\mathrm{HP},\mathrm{input}}\left(1-C_{\mathrm{HP}-\mathrm{loss}}\right)&\text{それ以外}\end{cases}\]

⑪ PD後確定クリティカル

\[U_{\mathrm{PDcrit}}=\mathcal C(f_{\mathrm{PD}},S_{\mathrm{PD}},L_{\mathrm{PDcrit}})\]
\[U_{\mathrm{PDcrit},x}=\begin{cases}U_{\mathrm{PDcrit}}&\text{装備効果として直接有効}\\U_{\mathrm{PDcrit}}\,U_{\mathrm{PS},x}&\text{PS側条件として扱う場合}\end{cases}\]
\[P_{m,x}=U_{\mathrm{PDcrit},x}+(1-U_{\mathrm{PDcrit},x})P^{(0)}_{m,x},\qquad P_{r,x}=U_{\mathrm{PDcrit},x}+(1-U_{\mathrm{PDcrit},x})P^{(0)}_{r,x}\]

⑬ AS静的CT・攻撃不能時間による待ち

\[C_0=\frac{C_{\mathrm{AS}}}{1+S_{\mathrm{AS}}},\qquad A=T_{\mathrm{attack}},\quad T=A+R,\quad W=\frac{R^2}{2T}\]
\[f_{\mathrm{base}}=\frac1{C_0+W}\]

⑭ 全攻撃クリティカル/弱点イベントとAS CT短縮

\[h_{\mathrm{all}}=h_{\mathrm{N}}+h_{\mathrm{AS}}+h_{\mathrm{ULT}}+h_{\mathrm{ES}}\]
\[P_{\mathrm{crit},\mathrm{all}}=\frac{h_{\mathrm{N}}P_{\mathrm{N}}+h_{\mathrm{AS}}P_{\mathrm{AS}}+h_{\mathrm{ULT}}P_{\mathrm{ULT}}+h_{\mathrm{ES}}P_{\mathrm{ES}}}{h_{\mathrm{all}}}\]
\[d_{\mathrm{crit}}=\frac{(d_{\mathrm{crit},\mathrm{M}}m_{\mathrm{AS}}+d_{\mathrm{crit},\mathrm{R}}(1-m_{\mathrm{AS}}))P_{\mathrm{crit},\mathrm{all}}}{N_{\mathrm{skill}}}\]
\[d_{\mathrm{weak}}=\frac{(d_{\mathrm{weak},\mathrm{M}}m_{\mathrm{AS}}+d_{\mathrm{weak},\mathrm{R}}(1-m_{\mathrm{AS}}))P_{\mathrm{weak}}}{N_{\mathrm{skill}}},\quad d_{\mathrm{G}}=\frac{d_{\mathrm{G}}^{(0)}U_{\mathrm{G}}}{N_{\mathrm{skill}}},\quad d_{\mathrm{PG}}=d_{\mathrm{PG}}^{(0)}f_{\mathrm{PG},\mathrm{success}}\]
\[d_{\mathrm{hit}}=d_{\mathrm{crit}}+d_{\mathrm{weak}}+d_{\mathrm{G}}+d_{\mathrm{PG}}\]

⑮ AS CT頻度・即時CT・実行時間

\[k=\frac{C_0^2d_{\mathrm{hit}}}{(C_0+W)^2},\qquad \lambda_0=P_{\mathrm{AS}}(h_{\mathrm{N}}+H_{\mathrm{AS}}f_{\mathrm{base}})\]
\[\lambda=\frac{\lambda_0}{1-\min(0.999,P_{\mathrm{AS}}H_{\mathrm{AS}}k)},\qquad f=f_{\mathrm{base}}+k\lambda\]
\[f_{\mathrm{AS},\mathrm{CT}}=\frac{f}{1-I_{\mathrm{AS}}},\qquad t_{\mathrm{AS}}=\frac{H_{\mathrm{AS}}}{r_{\mathrm{AS}}},\qquad f_{\mathrm{AS}}=\min\!\left(f_{\mathrm{AS},\mathrm{CT}},\frac1{t_{\mathrm{AS}}}\right)\]
\[h_{\mathrm{AS}}=f_{\mathrm{AS}}H_{\mathrm{AS}}\]

⑮-2 P3確率計算モード

\[K\sim\mathrm{Bin}(n,p)\]
\[p_{\mathrm{eff}}=\begin{cases} p & (\mathrm{expected})\\ \mathbb E\!\left[\dfrac{K}{n}\middle|K\in\mathcal T_{\mathrm{upper}}(Q)\right] & (\mathrm{upper})\\ \mathbb E\!\left[\dfrac{K}{n}\middle|K\in\mathcal T_{\mathrm{lower}}(Q)\right] & (\mathrm{lower}) \end{cases}\]
\[n_{\mathrm{crit},x}=h_xT_3,\qquad n_{\mathrm{instant}}=fT_3\]
\[P^{(0)}_{m,x}\to p_{\mathrm{eff}}\!\left(P^{(0)}_{m,x},n_{\mathrm{crit},m,x},Q\right),\qquad P^{(0)}_{r,x}\to p_{\mathrm{eff}}\!\left(P^{(0)}_{r,x},n_{\mathrm{crit},r,x},Q\right)\]
\[I_{\mathrm{AS}}\to p_{\mathrm{eff}}\!\left(I_{\mathrm{AS}},n_{\mathrm{instant}},Q\right)\]

⑯ ULT CT・ダウン/PG短縮・解放条件

\[S_{\mathrm{ULT},0,p}=S_{\mathrm{ULT}}+f_{\mathrm{PG},\mathrm{success},p}S_{\mathrm{PG},\mathrm{ULT}}\]
\[C_{\mathrm{ULT},0,p}=C_{\mathrm{ULT}}\frac{\max(1-S_{\mathrm{ULT},0,p},0)}{1+V_{\mathrm{ULT},p}},\qquad C_{\mathrm{ULT},\mathrm{down},p}=C_{\mathrm{ULT}}\frac{\max(1-S_{\mathrm{ULT},0,p}-S_{\mathrm{down}},0)}{1+V_{\mathrm{ULT},p}}\]
\[\Delta t_{0,p}=\max(t_{\mathrm{ULT}},C_{\mathrm{ULT},0,p}),\qquad \Delta t_{\mathrm{down},p}=\max(t_{\mathrm{ULT}},C_{\mathrm{ULT},\mathrm{down},p})\]
\[\text{1回のダウン短縮は1つのCT周期にのみ適用し、余剰短縮は次周期へ繰り越さない}\]
\[N_{\mathrm{ULT},3}=\min\!\left(N_{\mathrm{ULT,total}},\left\lfloor\frac{T_3}{t_{\mathrm{ULT}}}\right\rfloor\right),\qquad N_{\mathrm{ULT},2},N_{\mathrm{ULT},1}\text{ は残数を }P3\to P2\to P1\text{ の順で配分}\]

⑰ AS/ULT実行時間による通常攻撃占有

\[U_{\mathrm{AS}}=\min(1,f_{\mathrm{AS}}t_{\mathrm{AS}}),\quad U_{\mathrm{ULT}}=\min(1,f_{\mathrm{ULT}}t_{\mathrm{ULT}})\]
\[U_{\mathrm{skill}}=\min(1,U_{\mathrm{AS}}+U_{\mathrm{ULT}}),\qquad U_{\mathrm{N}}=\max(0,1-U_{\mathrm{skill}})\]

⑱ ES：チャージ型・CT型・バフ型

\[H_{\mathrm{need},\mathrm{eff}}=\frac{H_{\mathrm{need}}}{1+S_{\mathrm{EScharge}}},\qquad C_{\mathrm{ES},\mathrm{eff}}=\frac{C_{\mathrm{ES}}}{1+S_{\mathrm{EScharge}}}\]
\[f_{\mathrm{ES},\mathrm{charge}}\approx\frac{h_{\mathrm{trigger}}}{H_{\mathrm{need},\mathrm{eff}}},\qquad f_{\mathrm{ES},\mathrm{CT}}\approx\frac1{C_{\mathrm{ES},\mathrm{eff}}}\]
\[U_{\mathrm{ESbuff}}=\frac{T_{\mathrm{covered}}}{T_p},\qquad B^{\mathrm{eff}}_{\mathrm{ES}}=B_{\mathrm{ES}}U_{\mathrm{ESbuff}}\]

⑲ 1ヒット期待ダメージ

\[d_x=\mathrm{ATK}_x\,M_x\,C_{\mathrm{elem},x}\,C_{\mathrm{debuff}}\left[m_x C_{\mathrm{part},m,x}C_{\mathrm{crit},m,x}+(1-m_x)C_{\mathrm{part},r,x}C_{\mathrm{crit},r,x}\right]\]
\[C_{\mathrm{crit},m,x}=1+P_{m,x}\mathrm{CD}_x,\qquad C_{\mathrm{crit},r,x}=1+P_{r,x}\mathrm{CD}_x\]

⑳ 各攻撃DPS

\[\mathrm{DPS}_{\mathrm{N}}=r_{\mathrm{N}}\,d_{\mathrm{N}}\,U_{\mathrm{N}}\]
\[\mathrm{DPS}_{\mathrm{AS}}=h_{\mathrm{AS}}d_{\mathrm{AS}},\qquad \mathrm{DPS}_{\mathrm{ULT}}=h_{\mathrm{ULT}}d_{\mathrm{ULT}},\qquad \mathrm{DPS}_{\mathrm{ES}}=r_{\mathrm{ES},\mathrm{damage}}d_{\mathrm{ES}}\]

㉑ フェーズ総ダメージ・行動方策

\[D_p=T_p\sum_x \mathrm{DPS}_{x,p},\qquad D_{\mathrm{total}}=\sum_p D_p\]
\[\pi^*=\arg\max_{\pi\in\{\mathrm{AS},\mathrm{ULT},\mathrm{ES}\;\mathrm{use}/\mathrm{nonuse}\}}D_{\mathrm{total}}(E,\pi)\]

㉒ 装備探索

\[E^*=\arg\max_{E\in\mathcal E_{\mathrm{valid}}}D_{\mathrm{total}}(E,\pi^*(E))\]
